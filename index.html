// Temporizador de Descanso

function startRestTimer(seconds) {

clearInterval(restTimerInterval);

const timerBar = document.getElementById('restTimerBar');

const countdownEl = document.getElementById('timerCountdown');

timerBar.classList.remove('hidden');

let timeLeft = seconds;

updateTimerDisplay(timeLeft, countdownEl);

restTimerInterval = setInterval(() => {

timeLeft--;

updateTimerDisplay(timeLeft, countdownEl);

if (timeLeft <= 0) {

clearInterval(restTimerInterval);

timerBar.classList.add('hidden');

alert('¡Fin del descanso! A por la siguiente serie.');

}

}, 1000);

}

function updateTimerDisplay(seconds, el) {

const mins = Math.floor(seconds / 60).toString().padStart(2, '0');

const secs = (seconds % 60).toString().padStart(2, '0');

el.innerText = `${mins}:${secs}`;

}

// Calculadora de Nutrición Sin Pescado

function calculateNutrition() {

const weight = parseFloat(document.getElementById('userWeight').value) || 75;

const goal = document.getElementById('userGoal').value;

let baseCalories = weight * 33;

if (goal === 'hypertrophy') baseCalories += 350;

else baseCalories -= 400;

const proteinGrams = Math.round(weight * 2.1); // Alta proteína para hipertrofia

document.getElementById('resCalories').innerText = Math.round(baseCalories).toLocaleString();

document.getElementById('resProtein').innerText = `${proteinGrams}g`;

document.getElementById('resWater').innerText = `${(weight * 0.04).toFixed(1)}L`;

}

// Gráfica de Progreso y Peso

let weightChartInstance = null;

function initWeightChart() {

const ctx = document.getElementById('weightChart').getContext('2d');

const savedLogs = JSON.parse(localStorage.getItem('hyperforge_weight_logs')) || {

dates: ['Sem 1', 'Sem 2', 'Sem 3', 'Sem 4'],

weights: [74.2, 74.5, 74.9, 75.2]

};

weightChartInstance = new Chart(ctx, {

type: 'line',

data: {

labels: savedLogs.dates,

datasets: [{

label: 'Peso Corporal (kg)',

data: savedLogs.weights,

borderColor: '#22c55e',

backgroundColor: 'rgba(34, 197, 94, 0.1)',

borderWidth: 2,

tension: 0.3,

fill: true

}]

},

options: {

responsive: true,

maintainAspectRatio: false,

plugins: { legend: { display: false } },

scales: {

y: { grid: { color: '#1f2937' }, ticks: { color: '#9ca3af', font: { size: 10 } } },

x: { grid: { display: false }, ticks: { color: '#9ca3af', font: { size: 10 } } }

}

}

});

}

function saveWeightLog() {

const val = parseFloat(document.getElementById('logWeightInput').value);

if (!val) return;

const today = new Date().toLocaleDateString('es-ES', { month: 'short', day: 'numeric' });

let savedLogs = JSON.parse(localStorage.getItem('hyperforge_weight_logs')) || { dates: [], weights: [] };

savedLogs.dates.push(today);

savedLogs.weights.push(val);

localStorage.setItem('hyperforge_weight_logs', JSON.stringify(savedLogs));

weightChartInstance.data.labels = savedLogs.dates;

weightChartInstance.data.weights = savedLogs.weights;

weightChartInstance.update();

document.getElementById('logWeightInput').value = '';

alert('¡Peso guardado correctamente!');

}

// Lógica de PWA (Instalación en Android y PC)

let deferredPrompt;

window.addEventListener('beforeinstallprompt', (e) => {

e.preventDefault();

deferredPrompt = e;

const installBtn = document.getElementById('installBtn');

installBtn.classList.remove('hidden');

installBtn.addEventListener('click', async () => {

installBtn.classList.add('hidden');

deferredPrompt.prompt();

const { outcome } = await deferredPrompt.userChoice;

if (outcome === 'accepted') {

console.log('Usuario instaló la aplicación');

}

deferredPrompt = null;

});

});
